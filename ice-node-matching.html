# For each ice node, check if any roster facility shares city+state
ice_with_roster <- ice_nodes |>
  mutate(
    city_upper  = str_to_upper(str_trim(city)),
    state_upper = str_to_upper(str_trim(state))
  ) |>
  left_join(
    roster_by_loc |> 
      select(canonical_id, canonical_name, facility_address, facility_city,
             facility_state, detloc, facility_type_wiki, city_upper, state_upper),
    by = c("city_upper", "state_upper"),
    relationship = "many-to-many"
  )

# Now filter to cases where the street address actually matches
# (first word of street number and name)
matched <- ice_with_roster |>
  filter(!is.na(canonical_id)) |>
  mutate(
    node_street   = str_to_upper(str_trim(address_line1)),
    roster_street = str_to_upper(str_trim(facility_address)),
    # Check if the node street appears in the roster street or vice versa
    street_match = str_detect(roster_street, fixed(node_street, ignore_case = TRUE)) |
                   str_detect(node_street, fixed(roster_street, ignore_case = TRUE))
  ) |>
  filter(street_match)

not_matched <- ice_nodes |>
  dplyr::mutate(
    city_upper  = stringr::str_to_upper(stringr::str_trim(city)),
    state_upper = stringr::str_to_upper(stringr::str_trim(state)),
    node_street = stringr::str_to_upper(stringr::str_trim(address_line1))
    ) |>
  dplyr::filter(!node_id %in% matched$node_id)

# Cross-reference all 110 ice nodes against vera_facilities by state + street
vera_by_state <- vera_facilities |>
  dplyr::mutate(
    vera_street  = stringr::str_to_upper(stringr::str_extract(facility_address, "^[^,]+") |> stringr::str_trim()),
    state_upper  = stringr::str_to_upper(stringr::str_trim(facility_state))
  )

ice_nodes_vera <- ice_nodes |>
  dplyr::mutate(
    state_upper = stringr::str_to_upper(stringr::str_trim(state))
  ) |>
  dplyr::left_join(
    vera_by_state |>
      dplyr::select(detloc, facility_name, facility_address, facility_city, facility_state,
                    type_grouped_corrected, vera_street, state_upper),
    by = "state_upper",
    relationship = "many-to-many"
  ) |>
  dplyr::filter(!is.na(detloc)) |>
  dplyr::mutate(
    node_street  = stringr::str_to_upper(stringr::str_trim(address_line1)),

                   stringr::str_detect(node_street, stringr::fixed(vera_street, ignore_case = TRUE))
